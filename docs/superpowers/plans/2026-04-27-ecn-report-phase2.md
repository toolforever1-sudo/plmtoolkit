**Goal:** Build a standalone ECN Report tab with an interactive KPI dashboard, editable SLA targets with named profiles, and branded email delivery.

**Architecture:** Python generates raw ECN data (Excel + JSON sidecar). Java serves the JSON, manages SLA profiles/annotations/access control, and assembles emails. Browser renders Chart.js dashboard with instant client-side recalculation when targets change.

**Tech Stack:** Java 1.8 (Spring Boot), vanilla JavaScript, Chart.js 4.x (CDN), Python 3 (matplotlib for email chart PNGs), javax.mail for SMTP.

```java
package com.sandisk.plm.tracker.service;

import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.stereotype.Service;

import javax.annotation.PostConstruct;
import java.io.*;
import java.util.*;
import java.util.concurrent.ConcurrentHashMap;
import java.util.logging.Logger;

@Service
public class EcnReportService {

    private static final Logger logger = Logger.getLogger(EcnReportService.class.getName());
    private static final ObjectMapper mapper = new ObjectMapper();

    private static final String DATA_DIR = "./data/ecn-report";
    private static final String PROFILES_FILE = DATA_DIR + "/sla-profiles.json";
    private static final String ANNOTATIONS_FILE = DATA_DIR + "/annotations.json";
    private static final String DIST_LIST_FILE = DATA_DIR + "/distribution-list.json";
    private static final String ECN_DATA_FILE = DATA_DIR + "/ecn_data.json";

    // In-memory caches loaded from JSON files
    private Map<String, Object> profilesData = new LinkedHashMap<>();
    private Map<String, Object> annotationsData = new LinkedHashMap<>();
    private Map<String, Object> distListData = new LinkedHashMap<>();

    @PostConstruct
    public void init() {
        new File(DATA_DIR).mkdirs();
        new File(DATA_DIR + "/output").mkdirs();
        new File(DATA_DIR + "/output/charts").mkdirs();
        profilesData = loadJson(PROFILES_FILE);
        annotationsData = loadJson(ANNOTATIONS_FILE);
        distListData = loadJson(DIST_LIST_FILE);
        // Ensure default structure
        if (!profilesData.containsKey("profiles")) {
            profilesData.put("profiles", new ArrayList<>());
        }
        if (!profilesData.containsKey("editors")) {
            List<String> defaultEditors = new ArrayList<>();
            defaultEditors.add("jimmy.sessumes");
            defaultEditors.add("vikas.singh");
            profilesData.put("editors", defaultEditors);
            saveJson(PROFILES_FILE, profilesData);
        }
        if (!distListData.containsKey("recipients")) {
            distListData.put("recipients", new ArrayList<>());
            saveJson(DIST_LIST_FILE, distListData);
        }
    }

    // ── Role checks ──────────────────────────────────────────────────

    @SuppressWarnings("unchecked")
    public boolean isEditor(String username) {
        List<String> editors = (List<String>) profilesData.getOrDefault("editors", Collections.emptyList());
        return editors.contains(username);
    }

    public boolean canEdit(boolean isAdmin, String username) {
        return isAdmin || isEditor(username);
    }

    // ── Editors ──────────────────────────────────────────────────────

    @SuppressWarnings("unchecked")
    public List<String> getEditors() {
        return new ArrayList<>((List<String>) profilesData.getOrDefault("editors", Collections.emptyList()));
    }

    public synchronized void setEditors(List<String> editors) {
        profilesData.put("editors", new ArrayList<>(editors));
        saveJson(PROFILES_FILE, profilesData);
    }

    // ── Profiles ─────────────────────────────────────────────────────

    @SuppressWarnings("unchecked")
    public List<Map<String, Object>> getProfiles() {
        return new ArrayList<>((List<Map<String, Object>>) profilesData.getOrDefault("profiles", Collections.emptyList()));
    }

    @SuppressWarnings("unchecked")
    public synchronized Map<String, Object> saveProfile(Map<String, Object> profile) {
        String id = UUID.randomUUID().toString();
        profile.put("id", id);
        List<Map<String, Object>> profiles = (List<Map<String, Object>>) profilesData.get("profiles");
        profiles.add(profile);
        saveJson(PROFILES_FILE, profilesData);
        return profile;
    }

    @SuppressWarnings("unchecked")
    public synchronized boolean updateProfile(String id, Map<String, Object> updates) {
        List<Map<String, Object>> profiles = (List<Map<String, Object>>) profilesData.get("profiles");
        for (Map<String, Object> p : profiles) {
            if (id.equals(p.get("id"))) {
                if (updates.containsKey("name")) p.put("name", updates.get("name"));
                if (updates.containsKey("overrides")) p.put("overrides", updates.get("overrides"));
                saveJson(PROFILES_FILE, profilesData);
                return true;
            }
        }
        return false;
    }

    @SuppressWarnings("unchecked")
    public synchronized boolean deleteProfile(String id) {
        List<Map<String, Object>> profiles = (List<Map<String, Object>>) profilesData.get("profiles");
        boolean removed = profiles.removeIf(p -> id.equals(p.get("id")));
        if (removed) saveJson(PROFILES_FILE, profilesData);
        return removed;
    }

    @SuppressWarnings("unchecked")
    public synchronized boolean activateProfile(String id) {
        List<Map<String, Object>> profiles = (List<Map<String, Object>>) profilesData.get("profiles");
        boolean found = false;
        for (Map<String, Object> p : profiles) {
            if (id.equals(p.get("id"))) {
                p.put("active", true);
                found = true;
            } else {
                p.put("active", false);
            }
        }
        if (found) saveJson(PROFILES_FILE, profilesData);
        return found;
    }

    @SuppressWarnings("unchecked")
    public Map<String, Object> getActiveProfile() {
        List<Map<String, Object>> profiles = (List<Map<String, Object>>) profilesData.get("profiles");
        for (Map<String, Object> p : profiles) {
            if (Boolean.TRUE.equals(p.get("active"))) return p;
        }
        return null;
    }

    // ── Annotations ──────────────────────────────────────────────────

    public Map<String, Object> getAnnotations() {
        return new LinkedHashMap<>(annotationsData);
    }

    public synchronized void saveAnnotation(String ecnNumber, Map<String, Object> annotation) {
        annotationsData.put(ecnNumber, annotation);
        saveJson(ANNOTATIONS_FILE, annotationsData);
    }

    // ── Distribution List ────────────────────────────────────────────

    public Map<String, Object> getDistributionList() {
        return new LinkedHashMap<>(distListData);
    }

    public synchronized void setDistributionList(Map<String, Object> data) {
        distListData = new LinkedHashMap<>(data);
        saveJson(DIST_LIST_FILE, distListData);
    }

    // ── ECN Data ─────────────────────────────────────────────────────

    public Map<String, Object> getEcnData() {
        return loadJson(ECN_DATA_FILE);
    }

    public boolean hasEcnData() {
        return new File(ECN_DATA_FILE).exists();
    }

    // ── JSON helpers ─────────────────────────────────────────────────

    @SuppressWarnings("unchecked")
    private Map<String, Object> loadJson(String path) {
        File f = new File(path);
        if (!f.exists()) return new LinkedHashMap<>();
        try {
            return mapper.readValue(f, new TypeReference<LinkedHashMap<String, Object>>() {});
        } catch (Exception e) {
            logger.warning("Failed to load " + path + ": " + e.getMessage());
            return new LinkedHashMap<>();
        }
    }

    private void saveJson(String path, Map<String, Object> data) {
        try {
            mapper.writerWithDefaultPrettyPrinter().writeValue(new File(path), data);
        } catch (Exception e) {
            logger.warning("Failed to save " + path + ": " + e.getMessage());
        }
    }
}
```

```java
package com.sandisk.plm.tracker.controller;

import com.sandisk.plm.tracker.service.ActivityLogger;
import com.sandisk.plm.tracker.service.EcnReportService;
import com.sandisk.plm.tracker.service.EmailTemplateService;
import com.sandisk.plm.tracker.service.ReportService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.multipart.MultipartFile;

import javax.activation.DataHandler;
import javax.activation.DataSource;
import javax.mail.*;
import javax.mail.internet.*;
import javax.mail.util.ByteArrayDataSource;
import javax.servlet.http.HttpServletResponse;
import javax.servlet.http.HttpSession;
import java.io.*;
import java.nio.file.*;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.*;
import java.util.logging.Logger;

@RestController
@RequestMapping("/api/ecn-report")
public class EcnReportController {

    private static final Logger logger = Logger.getLogger(EcnReportController.class.getName());
    private static final String DATA_DIR = "./data/ecn-report";

    @Autowired private EcnReportService ecnService;
    @Autowired private ReportService reportService;
    @Autowired private ActivityLogger activityLogger;
    @Autowired private EmailTemplateService emailTemplate;

    @Value("${mail.smtp.host}") private String smtpHost;
    @Value("${mail.smtp.port:25}") private int smtpPort;
    @Value("${mail.from}") private String fromAddress;

    // ── Helper methods ───────────────────────────────────────────────

    private String username(HttpSession s) {
        Object v = s != null ? s.getAttribute("username") : null;
        return v != null ? v.toString() : "unknown";
    }

    private String displayName(HttpSession s) {
        Object v = s != null ? s.getAttribute("displayName") : null;
        return v != null ? v.toString() : username(s);
    }

    private boolean isAdmin(HttpSession s) {
        return Boolean.TRUE.equals(s.getAttribute("isPlmAdmin"));
    }

    private boolean canEdit(HttpSession s) {
        return isAdmin(s) || ecnService.isEditor(username(s));
    }

    private Map<String, Object> forbidden() {
        Map<String, Object> r = new LinkedHashMap<>();
        r.put("success", false);
        r.put("message", "You do not have permission for this action.");
        return r;
    }

    private Map<String, Object> ok(String msg) {
        Map<String, Object> r = new LinkedHashMap<>();
        r.put("success", true);
        r.put("message", msg);
        return r;
    }

    // ── 1. Run Report ────────────────────────────────────────────────

    @PostMapping("/run")
    public Map<String, Object> runReport(HttpSession session) {
        if (!canEdit(session)) return forbidden();

        String user = username(session);
        String display = displayName(session);
        String email = session.getAttribute("email") != null ? session.getAttribute("email").toString() : "";

        reportService.markBuiltinRunning("ecn-report-tab", "ECN Report");
        new Thread(() -> {
            try {
                String python = reportService.getPythonExe();
                ProcessBuilder pb = new ProcessBuilder(python,
                    "ecn_report_generator.py",
                    "--config", "ecn_report.properties",
                    "--output", "output/ECN_Report.xlsx",
                    "--json-output", "ecn_data.json",
                    "--no-email");
                pb.redirectErrorStream(true);
                pb.directory(new File(DATA_DIR));
                pb.environment().put("PYTHONIOENCODING", "utf-8");
                pb.environment().put("PLM_USER_EMAIL", email);

                Process proc = pb.start();
                try (BufferedReader br = new BufferedReader(new InputStreamReader(proc.getInputStream()))) {
                    String line;
                    while ((line = br.readLine()) != null) {
                        reportService.appendLog("ecn-report-tab", line);
                    }
                }
                int exitCode = proc.waitFor();
                if (exitCode == 0) {
                    reportService.markBuiltinCompleted("ecn-report-tab");
                    reportService.setOutputFile("ecn-report-tab", DATA_DIR + "/output/ECN_Report.xlsx");
                    logger.info("[ECN-REPORT-TAB] Completed successfully");
                } else {
                    reportService.markBuiltinFailed("ecn-report-tab", "Exit code " + exitCode);
                }
            } catch (Exception e) {
                reportService.markBuiltinFailed("ecn-report-tab", e.getMessage());
                logger.warning("[ECN-REPORT-TAB] Error: " + e.getMessage());
            }
        }).start();

        activityLogger.log(user, display, "ECN_REPORT_RUN", "Triggered ECN Report generation");
        return ok("ECN Report generation started.");
    }

    // ── 2. Get Data ──────────────────────────────────────────────────

    @GetMapping("/data")
    public Map<String, Object> getData(HttpSession session) {
        Map<String, Object> result = new LinkedHashMap<>();
        if (!ecnService.hasEcnData()) {
            result.put("success", false);
            result.put("message", "No ECN data available. Run the report first.");
            return result;
        }
        Map<String, Object> ecnData = ecnService.getEcnData();
        Map<String, Object> activeProfile = ecnService.getActiveProfile();
        result.put("success", true);
        result.put("ecnData", ecnData);
        result.put("activeProfile", activeProfile);
        result.put("annotations", ecnService.getAnnotations());
        result.put("canEdit", canEdit(session));
        result.put("isAdmin", isAdmin(session));
        return result;
    }

    // ── 3. Download Excel ────────────────────────────────────────────

    @GetMapping("/download")
    public void downloadExcel(HttpSession session, HttpServletResponse response) throws IOException {
        File file = new File(DATA_DIR + "/output/ECN_Report.xlsx");
        if (!file.exists()) {
            response.setStatus(404);
            response.setContentType("application/json");
            response.getWriter().write("{\"error\":\"No report file available\"}");
            return;
        }
        response.setContentType("application/vnd.openxmlformats-officedocument.spreadsheetml.sheet");
        response.setHeader("Content-Disposition", "attachment; filename=\"ECN_Report.xlsx\"");
        Files.copy(file.toPath(), response.getOutputStream());
        activityLogger.log(username(session), displayName(session), "ECN_REPORT_DOWNLOAD", "Downloaded ECN Report Excel");
    }

    // ── 4. Upload SLA Template ───────────────────────────────────────

    @PostMapping("/upload-sla")
    public Map<String, Object> uploadSla(@RequestParam("file") MultipartFile file, HttpSession session) {
        if (!canEdit(session)) return forbidden();
        try {
            String filename = "ecn-kpi-template.xlsx";
            File dest = new File(DATA_DIR + "/" + filename);
            file.transferTo(dest);
            activityLogger.log(username(session), displayName(session), "ECN_SLA_UPLOAD",
                "Uploaded new SLA template: " + file.getOriginalFilename());
            return ok("SLA template uploaded successfully.");
        } catch (Exception e) {
            Map<String, Object> r = new LinkedHashMap<>();
            r.put("success", false);
            r.put("message", "Upload failed: " + e.getMessage());
            return r;
        }
    }

    // ── 5-9. Profile CRUD + Activate ─────────────────────────────────

    @GetMapping("/profiles")
    public List<Map<String, Object>> listProfiles(HttpSession session) {
        return ecnService.getProfiles();
    }

    @PostMapping("/profiles")
    public Map<String, Object> createProfile(@RequestBody Map<String, Object> body, HttpSession session) {
        // Viewers can save personal profiles, editors/admins can save any
        String user = username(session);
        body.put("createdBy", user);
        body.put("createdAt", LocalDateTime.now().format(DateTimeFormatter.ISO_LOCAL_DATE_TIME));
        body.put("active", false);
        Map<String, Object> saved = ecnService.saveProfile(body);
        activityLogger.log(user, displayName(session), "ECN_PROFILE_CREATE",
            "Created SLA profile: " + body.get("name"));
        Map<String, Object> result = ok("Profile saved.");
        result.put("profile", saved);
        return result;
    }

    @PutMapping("/profiles/{id}")
    public Map<String, Object> updateProfile(@PathVariable String id,
            @RequestBody Map<String, Object> body, HttpSession session) {
        // Check ownership for viewers
        if (!canEdit(session)) {
            // Viewers can only update own profiles
            List<Map<String, Object>> profiles = ecnService.getProfiles();
            boolean isOwner = profiles.stream()
                .anyMatch(p -> id.equals(p.get("id")) && username(session).equals(p.get("createdBy")));
            if (!isOwner) return forbidden();
        }
        if (ecnService.updateProfile(id, body)) {
            return ok("Profile updated.");
        }
        Map<String, Object> r = new LinkedHashMap<>();
        r.put("success", false);
        r.put("message", "Profile not found.");
        return r;
    }

    @DeleteMapping("/profiles/{id}")
    public Map<String, Object> deleteProfile(@PathVariable String id, HttpSession session) {
        if (!canEdit(session)) {
            List<Map<String, Object>> profiles = ecnService.getProfiles();
            boolean isOwner = profiles.stream()
                .anyMatch(p -> id.equals(p.get("id")) && username(session).equals(p.get("createdBy")));
            if (!isOwner) return forbidden();
        }
        if (ecnService.deleteProfile(id)) {
            return ok("Profile deleted.");
        }
        Map<String, Object> r = new LinkedHashMap<>();
        r.put("success", false);
        r.put("message", "Profile not found.");
        return r;
    }

    @PutMapping("/profiles/{id}/activate")
    public Map<String, Object> activateProfile(@PathVariable String id, HttpSession session) {
        if (!canEdit(session)) return forbidden();
        if (ecnService.activateProfile(id)) {
            activityLogger.log(username(session), displayName(session), "ECN_PROFILE_ACTIVATE",
                "Promoted SLA profile to active: " + id);
            return ok("Profile activated as the new SLA.");
        }
        Map<String, Object> r = new LinkedHashMap<>();
        r.put("success", false);
        r.put("message", "Profile not found.");
        return r;
    }

    // ── 10. Recalculate ──────────────────────────────────────────────

    @PostMapping("/recalculate")
    public Map<String, Object> recalculate(@RequestBody Map<String, Object> body, HttpSession session) {
        if (!canEdit(session)) return forbidden();
        // The recalculation is done client-side for preview; this endpoint
        // persists the profile and could regenerate the Excel in the future.
        // For now, it saves the profile if a profileId is provided.
        String profileId = (String) body.get("profileId");
        if (profileId != null) {
            Map<String, Object> overrides = new LinkedHashMap<>();
            overrides.put("overrides", body.get("overrides"));
            ecnService.updateProfile(profileId, overrides);
        }
        activityLogger.log(username(session), displayName(session), "ECN_RECALCULATE",
            "Server-side recalculation triggered");
        return ok("Recalculation complete.");
    }

    // ── 11. Send Email ───────────────────────────────────────────────

    @SuppressWarnings("unchecked")
    @PostMapping("/email")
    public Map<String, Object> sendEmail(@RequestBody Map<String, Object> body, HttpSession session) {
        if (!canEdit(session)) return forbidden();

        List<String> recipients = (List<String>) body.get("recipients");
        String subject = (String) body.getOrDefault("subject",
            "ECN Report — YTD (as of " + LocalDate.now().format(DateTimeFormatter.ofPattern("MMM dd, yyyy")) + ")");
        String htmlContent = (String) body.get("htmlContent");

        if (recipients == null || recipients.isEmpty()) {
            Map<String, Object> r = new LinkedHashMap<>();
            r.put("success", false);
            r.put("message", "No recipients specified.");
            return r;
        }

        try {
            Properties props = new Properties();
            props.put("mail.smtp.host", smtpHost);
            props.put("mail.smtp.port", String.valueOf(smtpPort));
            javax.mail.Session mailSession = javax.mail.Session.getInstance(props);

            // Read the Excel attachment
            File excelFile = new File(DATA_DIR + "/output/ECN_Report.xlsx");
            byte[] excelBytes = excelFile.exists() ? Files.readAllBytes(excelFile.toPath()) : null;

            for (String recipientEmail : recipients) {
                MimeMessage message = new MimeMessage(mailSession);
                message.setFrom(new InternetAddress(fromAddress));
                message.setRecipient(Message.RecipientType.TO, new InternetAddress(recipientEmail.trim()));
                message.setSubject(subject);

                if (excelBytes != null && htmlContent != null) {
                    // Multipart: HTML body + Excel attachment
                    MimeBodyPart textPart = new MimeBodyPart();
                    textPart.setContent(htmlContent, "text/html; charset=utf-8");

                    MimeBodyPart attachPart = new MimeBodyPart();
                    DataSource ds = new ByteArrayDataSource(excelBytes,
                        "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet");
                    attachPart.setDataHandler(new DataHandler(ds));
                    attachPart.setFileName("ECN_Report.xlsx");

                    // Check for chart images and attach as inline CID
                    Multipart multipart = new MimeMultipart("related");
                    multipart.addBodyPart(textPart);
                    multipart.addBodyPart(attachPart);

                    File chartsDir = new File(DATA_DIR + "/output/charts");
                    if (chartsDir.exists()) {
                        File[] chartFiles = chartsDir.listFiles((dir, name) -> name.endsWith(".png"));
                        if (chartFiles != null) {
                            for (File chart : chartFiles) {
                                MimeBodyPart chartPart = new MimeBodyPart();
                                chartPart.attachFile(chart);
                                chartPart.setContentID("<" + chart.getName().replace(".png", "") + ">");
                                chartPart.setDisposition(MimeBodyPart.INLINE);
                                multipart.addBodyPart(chartPart);
                            }
                        }
                    }

                    message.setContent(multipart);
                } else if (htmlContent != null) {
                    message.setContent(htmlContent, "text/html; charset=utf-8");
                } else {
                    message.setText("ECN Report — see attached Excel file.");
                }

                Transport.send(message);
                logger.info("[ECN-REPORT] Email sent to " + recipientEmail);
            }

            activityLogger.log(username(session), displayName(session), "ECN_REPORT_EMAIL",
                "Sent ECN Report email to " + recipients.size() + " recipients");
            return ok("Email sent to " + recipients.size() + " recipient(s).");

        } catch (Exception e) {
            logger.warning("[ECN-REPORT] Email failed: " + e.getMessage());
            Map<String, Object> r = new LinkedHashMap<>();
            r.put("success", false);
            r.put("message", "Email failed: " + e.getMessage());
            return r;
        }
    }

    // ── 12-13. Editors ───────────────────────────────────────────────

    @GetMapping("/editors")
    public Map<String, Object> getEditors(HttpSession session) {
```

```javascript
// ── Target Merging ───────────────────────────────────────────────────

function ecnMergeTargets(baseline, profile) {
    var merged = {};
    // Copy all baseline targets
    for (var key in baseline) {
        merged[key] = { standard: baseline[key].standard, urgent: baseline[key].urgent };
    }
    // Apply profile overrides
    if (profile && profile.overrides) {
        for (var key in profile.overrides) {
            if (merged[key]) {
                if (profile.overrides[key].standard !== undefined) merged[key].standard = profile.overrides[key].standard;
                if (profile.overrides[key].urgent !== undefined) merged[key].urgent = profile.overrides[key].urgent;
            } else {
                merged[key] = { standard: profile.overrides[key].standard, urgent: profile.overrides[key].urgent };
            }
        }
    }
    // Apply dirty (unsaved) edits on top
    for (var key in ecnState.dirtyTargets) {
        if (merged[key]) {
            if (ecnState.dirtyTargets[key].standard !== undefined) merged[key].standard = ecnState.dirtyTargets[key].standard;
            if (ecnState.dirtyTargets[key].urgent !== undefined) merged[key].urgent = ecnState.dirtyTargets[key].urgent;
        }
    }
    return merged;
}

function ecnComputeStats() {
    var ecns = ecnState.ecns;
    var targets = ecnState.currentTargets;
    var annotations = ecnState.annotations;

    // Apply annotation overrides to ECN data
    var rows = ecns.map(function(e) {
        var a = annotations[e.number];
        var row = {};
        for (var k in e) row[k] = e[k];
        if (a) {
            if (a.teamOverride) row.productTeam = a.teamOverride;
            if (a.classificationOverride) row.ecnClassification = a.classificationOverride;
        }
        // Compute target key and on-target status
        var tKey = (row.requestClassification || '') + '|' + (row.requestSubclass || '');
        var t = targets[tKey];
        row._targetKey = tKey;
        if (t) {
            var targetDays = row.priority === 'Urgent' ? t.urgent : t.standard;
            row._targetDays = targetDays;
            if (targetDays === 'NA' || targetDays === 'TBD' || targetDays === null || targetDays === undefined) {
                row._onTarget = null;
            } else {
                row._onTarget = row.actualDays <= parseFloat(targetDays);
            }
        } else {
            row._targetDays = null;
            row._onTarget = null;
        }
        return row;
    });

    // Filter to completed only for KPI stats
    var completed = rows.filter(function(r) { return r.status === 'Completed'; });

    var stats = {};

    // ── ECN Volume YTD ───────────────────────────────────────────
    stats.volume = ecnComputeVolumePanel(completed);

    // ── Cycle Time (General ECN) YTD ─────────────────────────────
    stats.ctGeneral = ecnComputeCycleTimePanel(
        completed.filter(function(r) { return r.ecnClassification === 'General ECNs'; }),
        'General ECN'
    );

    // ── Cycle Time (PDR ECN) YTD ─────────────────────────────────
    stats.ctPdr = ecnComputeCycleTimePanel(
        completed.filter(function(r) { return (r.ecnClassification || '').indexOf('PDR') >= 0; }),
        'PDR ECN'
    );

    // ── ECN Volume by Change Type YTD ────────────────────────────
    stats.volumeByType = ecnComputeVolumeByType(completed);

    // ── Cycle Time (Dedicated Process) YTD ───────────────────────
    stats.ctDedicated = ecnComputeCycleTimePanel(
        completed.filter(function(r) { return (r.ecnClassification || '').indexOf('Dedicated') >= 0; }),
        'Dedicated Process ECN'
    );

    // ── Cycle Time by Product Team YTD ───────────────────────────
    stats.ctTeam = ecnComputeTeamPanel(completed);

    stats.trends = ecnComputeTrends(completed);

function ecnComputeVolumePanel(ecns) {
    var total = ecns.length;
    var standard = ecns.filter(function(r) { return r.priority === 'Standard'; });
    var urgent = ecns.filter(function(r) { return r.priority === 'Urgent'; });
    return {
        total: total,
        standard: { count: standard.length, pct: total ? Math.round(standard.length / total * 100) : 0 },
        urgent: { count: urgent.length, pct: total ? Math.round(urgent.length / total * 100) : 0 }
    };
}

function ecnComputeCycleTimePanel(ecns, label) {
    var total = ecns.length;
    var standard = ecns.filter(function(r) { return r.priority === 'Standard'; });
    var urgent = ecns.filter(function(r) { return r.priority === 'Urgent'; });

    function avg(arr) {
        if (arr.length === 0) return 0;
        var sum = 0;
        for (var i = 0; i < arr.length; i++) sum += (arr[i].actualDays || 0);
        return Math.round(sum / arr.length * 10) / 10;
    }

    function pctOnTarget(arr) {
        var trackable = arr.filter(function(r) { return r._onTarget !== null; });
        if (trackable.length === 0) return null;
        var onTarget = trackable.filter(function(r) { return r._onTarget; }).length;
        return Math.round(onTarget / trackable.length * 100);
    }

    return {
        label: label,
        total: total,
        pctOfOverall: null, // filled in by caller if needed
        avgDays: avg(ecns),
        pctOnTarget: pctOnTarget(ecns),
        standard: {
            count: standard.length,
            pct: total ? Math.round(standard.length / total * 100) : 0,
            avgDays: avg(standard),
            pctOnTarget: pctOnTarget(standard)
        },
        urgent: {
            count: urgent.length,
            pct: total ? Math.round(urgent.length / total * 100) : 0,
            avgDays: avg(urgent),
            pctOnTarget: pctOnTarget(urgent)
        }
    };
}

function ecnComputeVolumeByType(ecns) {
    var groups = {};
    for (var i = 0; i < ecns.length; i++) {
        var ct = ecns[i].changeType || 'Unknown';
        if (!groups[ct]) groups[ct] = [];
        groups[ct].push(ecns[i]);
    }
    var total = ecns.length;
    var result = [];
    for (var ct in groups) {
        var arr = groups[ct];
        var sum = 0;
        for (var j = 0; j < arr.length; j++) sum += (arr[j].actualDays || 0);
        result.push({
            changeType: ct,
            count: arr.length,
            pct: total ? Math.round(arr.length / total * 100) : 0,
            avgDays: arr.length ? Math.round(sum / arr.length * 10) / 10 : 0
        });
    }
    result.sort(function(a, b) { return b.count - a.count; });
    return result;
}

function ecnComputeTeamPanel(ecns) {

            var t = arr.filter(function(r) { return r._onTarget !== null; });
            if (!t.length) return 0;
            return Math.round(t.filter(function(r) { return r._onTarget; }).length / t.length * 100);
        }

        trends[m.key] = {
            label: m.label,
            total: monthEcns.length,
            standard: standard.length,
            urgent: urgent.length,
            avgDays: avgDays(monthEcns),
            pctOnTarget: pctOnTarget(monthEcns),
            avgDaysStd: avgDays(standard),
            avgDaysUrg: avgDays(urgent),
            pctOnTargetStd: pctOnTarget(standard),
            pctOnTargetUrg: pctOnTarget(urgent)
        };
    }
    return { months: months, data: trends };
}

function ecnGetLast3Months() {
    var end = ecnState.dateRange ? new Date(ecnState.dateRange.end) : new Date();
    var months = [];
    for (var i = 2; i >= 0; i--) {
        var d = new Date(end.getFullYear(), end.getMonth() - i, 1);
        var key = d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0');
        var label = d.toLocaleString('en-US', { month: 'short', year: 'numeric' });
        months.push({ key: key, label: label });
    }
    return months;
}
```

```javascript
function ecnSaveAndRecalculate() {
    if (!ecnState.activeProfile) {
        ecnSaveProfileModal();
        return;
    }
    // Merge dirty targets into the active profile's overrides
    var overrides = ecnState.activeProfile.overrides || {};
    for (var key in ecnState.dirtyTargets) {
        if (!overrides[key]) overrides[key] = {};
        for (var p in ecnState.dirtyTargets[key]) {
            overrides[key][p] = ecnState.dirtyTargets[key][p];
        }
    }
    fetch('/api/ecn-report/recalculate', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ profileId: ecnState.activeProfile.id, overrides: overrides })
    })
    .then(function(r) { return r.json(); })
    .then(function(data) {
        if (data.success) {
            ecnState.activeProfile.overrides = overrides;
            ecnState.dirtyTargets = {};
            ecnRefreshUI();
        } else {
            alert(data.message || 'Recalculation failed.');
```

```javascript
function ecnRunReport() {

    document.getElementById('ecnProgressWrap').style.display = '';

var ecnPollTimer = null;

function ecnPollStatus() {
    if (ecnPollTimer) clearInterval(ecnPollTimer);
    var startTime = Date.now();
    ecnPollTimer = setInterval(function() {
        fetch('/api/ecn-report/status')
            .then(function(r) { return r.json(); })
            .then(function(data) {
                var elapsed = Math.round((Date.now() - startTime) / 1000);
                if (data.status === 'running') {
                    var pct = Math.min(90, elapsed * 3); // rough estimate
                    document.getElementById('ecnProgressBar').style.width = pct + '%';
                    document.getElementById('ecnProgressText').textContent = 'Running... ' + elapsed + 's elapsed';
                } else if (data.status === 'completed') {
                    clearInterval(ecnPollTimer);
                    ecnPollTimer = null;
                    document.getElementById('ecnProgressBar').style.width = '100%';
                    document.getElementById('ecnProgressText').textContent = 'Complete! Loading data...';
                    ecnState.loaded = false;
                    ecnFetchData();
                    setTimeout(function() {
                        document.getElementById('ecnProgressWrap').style.display = 'none';
                    }, 2000);
                } else if (data.status === 'failed') {
                    clearInterval(ecnPollTimer);
                    ecnPollTimer = null;
                    document.getElementById('ecnProgressBar').style.width = '100%';
                    document.getElementById('ecnProgressBar').style.background = '#B8342B';
                    document.getElementById('ecnProgressText').textContent = 'Failed: ' + (data.error || 'unknown error');
                }
            })
            .catch(function() {});

// ── Manage Editors Modal ─────────────────────────────────────────────

function ecnShowEditorsModal() {
    fetch('/api/ecn-report/editors')
        .then(function(r) { return r.json(); })
        .then(function(data) {
            if (!data.success) { alert(data.message); return; }
            var editors = data.editors || [];
            var html = '<div id="ecnEditorsModal" style="position:fixed; top:0; left:0; right:0; bottom:0; background:rgba(0,0,0,0.4); z-index:2000; display:flex; align-items:center; justify-content:center;">' +
                '<div style="background:#fff; border-radius:8px; padding:24px; max-width:400px; width:90%;">' +
                '<h3 style="margin:0 0 12px;">Manage ECN Report Editors</h3>' +
                '<div id="ecnEditorsList">';
            for (var i = 0; i < editors.length; i++) {
                html += '<div style="display:flex; align-items:center; padding:4px 0;">' +
                    '<span style="flex:1; font-size:13px;">' + editors[i] + '</span>' +
                    '<button class="tool-btn" style="font-size:11px; color:#B8342B;" onclick="ecnRemoveEditor(\'' + editors[i] + '\')">Remove</button>' +
                    '</div>';
            }
            html += '</div>' +
                '<div style="display:flex; gap:8px; margin-top:12px;">' +
                '<input type="text" id="ecnNewEditor" placeholder="username" style="flex:1; padding:5px 8px; border:1px solid #ccc; border-radius:4px; font-size:13px;">' +
                '<button class="tool-btn" onclick="ecnAddEditor()">+ Add</button>' +
                '</div>' +
                '<div style="text-align:right; margin-top:16px;">' +
                '<button class="tool-btn" onclick="ecnCloseEditorsModal()">Done</button>' +
                '</div></div></div>';
            document.body.insertAdjacentHTML('beforeend', html);
        })
        .catch(function(err) { alert('Error: ' + err.message); });
}

function ecnCloseEditorsModal() {
    var m = document.getElementById('ecnEditorsModal');
    if (m) m.remove();
}

function ecnRemoveEditor(username) {
    ecnUpdateEditorsList(function(editors) {
        return editors.filter(function(e) { return e !== username; });
    });
}

function ecnAddEditor() {
    var input = document.getElementById('ecnNewEditor');
    var username = input.value.trim();
    if (!username) return;
    ecnUpdateEditorsList(function(editors) {
        if (editors.indexOf(username) < 0) editors.push(username);
        return editors;
    });
    input.value = '';
}

function ecnUpdateEditorsList(transform) {
    fetch('/api/ecn-report/editors')
        .then(function(r) { return r.json(); })
        .then(function(data) {
            var editors = transform(data.editors || []);
            return fetch('/api/ecn-report/editors', {
                method: 'PUT',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ editors: editors })
            });
        })
        .then(function(r) { return r.json(); })
        .then(function(data) {
            if (data.success) {
                ecnCloseEditorsModal();
                ecnShowEditorsModal(); // re-render
            } else {
                alert(data.message || 'Failed to update editors.');
            }
        })
        .catch(function(err) { alert('Error: ' + err.message); });
}
```
